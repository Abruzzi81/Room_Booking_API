async function test(name, method, endpoint, token, body, expected) {
  try {
    const headers = {};

    if (token) {
      headers.Authorization = `Bearer ${token}`;
    }

    if (body) {
      headers['Content-Type'] = 'application/json';
    }

    const response = await fetch(`${BASE_URL}${endpoint}`, {
      method,
      headers,
      body: body ? JSON.stringify(body) : undefined
    });

    const text = await response.text();

    let data;
    try {
      data = JSON.parse(text);
    } catch {
      data = text;
    }

    const passed = expected.includes(response.status);

    const result = {
      name,
      status: response.status,
      expected: expected.join(' lub '),
      passed,
      details: typeof data === 'object'
        ? JSON.stringify(data)
        : data
    };

    results.push(result);

    console.log(
      `${passed ? 'PASS' : 'FAIL'} | ${name} | HTTP ${response.status} | oczekiwano: ${result.expected}`
    );

    return data;
  } catch (error) {
    results.push({
      name,
      status: 'ERROR',
      expected: expected.join(' lub '),
      passed: false,
      details: error.message
    });

    console.log(`ERROR | ${name} | ${error.message}`);
    return null;

const express = require('express');

const errorMiddleware = require('./middleware/error.middleware');
const NotFoundError = require('./errors/not-found.error');

const roomsRoutes = require('./modules/rooms/rooms.routes');
const reservationsRoutes = require('./modules/reservations/reservations.routes');
const buildingsRoutes = require('./modules/buildings/buildings.routes');

const authRoutes = require('./modules/auth/auth.routes');
const { authenticate } = require('./middleware/auth.middleware');


function createApp() {
  const app = express();

  app.use(express.json());
  app.use(loggerMiddleware); // Logowanie i X-Request-Id

  app.use((req, res, next) => {
    const publicRoutes = [
      req.method === 'POST' && req.path === '/api/auth/register',
      req.method === 'POST' && req.path === '/api/auth/login',
      req.path === '/api/docs' || req.path.startsWith('/api/docs/'),
    ];

    if (publicRoutes.some(Boolean)) {
      return next();
    }

    return authenticate(req, res, next);
  });

  // Trasy API
  app.use('/api/rooms', roomsRoutes);
  app.use('/api/reservations', reservationsRoutes);
  app.use('/api/buildings', buildingsRoutes);

  app.use('/api/auth', authRoutes);

  // ObsĹ‚uga nieistniejÄ…cych Ĺ›cieĹĽek (404 w formacie JSON)
  app.use((req, res, next) => {
    next(new NotFoundError(`Route ${req.originalUrl} not found`));
  });

  // Centralny error handler (musi byÄ‡ na samym koĹ„cu)
  app.use(errorMiddleware);

  return app;
}

module.exports = { createApp };

function authenticate(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader) {
    return res.status(401).json({
      code: 'UNAUTHORIZED',
      message: 'Brak tokenu',
    });
  }

  if (
    !authHeader.startsWith('Bearer ') ||
    !authHeader.slice(7).trim()
  ) {
    return res.status(401).json({
      code: 'UNAUTHORIZED',
      message: 'NieprawidĹ‚owy format nagĹ‚Ăłwka Authorization',
    });
  }

  const token = authHeader.slice(7).trim();
  const secret = process.env.JWT_SECRET;

  if (!secret) {
    return next(new Error('Brak konfiguracji JWT_SECRET'));
  }

  try {
    const payload = jwt.verify(token, secret, {
      algorithms: ['HS256'],
    });

    if (
      typeof payload === 'string' ||
      typeof payload.sub !== 'string' ||
      !['USER', 'ADMIN'].includes(payload.role)
    ) {
      return res.status(401).json({
        code: 'UNAUTHORIZED',
        message: 'NieprawidĹ‚owy token',
      });
    }

    req.user = {
      id: payload.sub,
      role: payload.role,
    };

    return next();
  } catch (error) {
    if (error.name === 'TokenExpiredError') {
      return res.status(401).json({
        code: 'TOKEN_EXPIRED',
        message: 'Token expired',
      });
    }

    return res.status(401).json({
      code: 'UNAUTHORIZED',
      message: 'NieprawidĹ‚owy lub zmodyfikowany token',
    });
  }
}

function authorize(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return res.status(403).json({
        code: 'FORBIDDEN',
        message: 'Brak uprawnieĹ„ do wykonania tej operacji',
      });
    }

    return next();
  };
}

module.exports = { authenticate, authorize };

const AppError = require('../errors/app.error');
const { ZodError } = require('zod');

function errorMiddleware(err, req, res, next) {
  const requestId = req.requestId || 'unknown';

  // ObsĹ‚uga bĹ‚Ä™dĂłw parsowania JSON-a przez Express (body-parser)
  if (err instanceof SyntaxError && err.status === 400 && 'body' in err) {
    return res.status(400).json({
      error: {
        code: "INVALID_JSON",
        message: "Invalid JSON format in request body",
        requestId
      }
    });
  }

  // ObsĹ‚uga bĹ‚Ä™dĂłw walidacji Zod (Zapewnia status 400 zamiast 500)
  if (err instanceof ZodError || err.name === 'ZodError') {
    const details = (err.errors || err.issues || []).map((e) => ({
      location: e.path[0] || 'body',
      field: e.path.slice(1).join('.') || e.path[e.path.length - 1] || '',
      message: e.message,
    }));

    return res.status(400).json({
      error: {
        code: "VALIDATION_ERROR",
        message: "Validation failed",
        details,
        requestId
      }
    });
  }

  // ObsĹ‚uga zdefiniowanych bĹ‚Ä™dĂłw aplikacji (AppError i dziedziczÄ…ce)
  if (err instanceof AppError) {
    const responsePayload = {
      error: {
        code: err.code,
        message: err.message,
        requestId
      }
    };
    if (err.details) {
      responsePayload.error.details = err.details;
    }
    return res.status(err.statusCode).json(responsePayload);
  }

  // ObsĹ‚uga bĹ‚Ä™dĂłw bazy danych PostgreSQL (np. Unique Violation 23505)
  if (err.code === '23505') {
    let message = 'Resource already exists';
    let code = 'CONFLICT';

    // RozrĂłĹĽnienie constraintĂłw bez wyciekania wewnÄ™trznej struktury bazy do klienta
    if (err.constraint && err.constraint.includes('rooms')) {
      message = 'A room with this name already exists in this building';
      code = 'ROOM_ALREADY_EXISTS';
    } else if (err.constraint && err.constraint.includes('buildings')) {
      message = 'A building with this name already exists';
      code = 'BUILDING_ALREADY_EXISTS';
    }

    return res.status(409).json({
      error: {
        code,
        message,
        requestId
      }
    });
  }


  // Nieoczekiwany bĹ‚Ä…d serwera (500) - stack trace trafia tylko do logĂłw serwera
  console.error(`[CRITICAL ERROR] ${requestId}:`, err);

  return res.status(500).json({
    error: {
      code: "INTERNAL_SERVER_ERROR",
      message: "An unexpected error occurred",
      requestId
    }
  });
}

module.exports = errorMiddleware;

const crypto = require('crypto');

function loggerMiddleware(req, res, next) {
  const requestId = req.headers['x-request-id'] || crypto.randomUUID();
  req.requestId = requestId;
  res.setHeader('X-Request-Id', requestId);

  const start = process.hrtime();

  res.on('finish', () => {
    const [seconds, nanoseconds] = process.hrtime(start);
    const durationMs = (seconds * 1000 + nanoseconds / 1e6).toFixed(2);
    
    console.log(`[${new Date().toISOString()}] ${requestId} - ${req.method} ${req.originalUrl} - Status: ${res.statusCode} - ${durationMs}ms`);
  });

  next();
}

module.exports = loggerMiddleware;

const validate = (schema) => async (req, res, next) => {
  try {
    const parsed = await schema.parseAsync({
      body: req.body,
      query: req.query,
      params: req.params,
    });
    
    // Przypisanie zweryfikowanych danych do req.validated
    req.validated = parsed;
    return next();
  } catch (error) {
    // Sprawdzamy, czy to bĹ‚Ä…d Zoda i czy zawiera tablicÄ™ bĹ‚Ä™dĂłw/issues
    if (error instanceof ZodError || error.name === 'ZodError') {
      const issues = error.errors || error.issues || [];
      
      const details = issues.map((err) => ({
        location: err.path[0] || 'body',
        field: err.path.slice(1).join('.') || err.path[err.path.length - 1] || '',
        message: err.message,
      }));

      return next(new ValidationError("Validation failed", details));
    }

    // JeĹ›li to inny bĹ‚Ä…d, przekazujemy go dalej
    return next(error);
  }
};

const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

const { registerSchema, loginSchema } = require('./auth.schema');
const usersRepository = require('./users.repository.knex');
const { toUserResponse } = require('./users.mapper');

function createError(code, message) {
  const error = new Error(message);
  error.code = code;
  return error;
}

async function register(data) {
  const result = registerSchema.safeParse(data);

  if (!result.success) {
    throw createError(
      'VALIDATION_ERROR',
      'NieprawidĹ‚owe dane rejestracji'
    );
  }

  const { email, password } = result.data;
  const passwordHash = await bcrypt.hash(password, 12);

  try {
    const user = await usersRepository.create({
      email,
      passwordHash,
      role: 'USER',
    });

    return toUserResponse(user);
  } catch (error) {
    if (
      error.code === '23505' &&
      error.constraint === 'users_email_unique'
    ) {
      throw createError('EMAIL_TAKEN', 'Email jest juĹĽ zajÄ™ty');
    }

    throw error;
  }
}

async function login(data) {
  const result = loginSchema.safeParse(data);

  if (!result.success) {
    throw createError(
      'INVALID_CREDENTIALS',
      'NieprawidĹ‚owy email lub hasĹ‚o'
    );
  }

  const { email, password } = result.data;
  const user = await usersRepository.findByEmail(email);

  if (!user) {
    throw createError(
      'INVALID_CREDENTIALS',
      'NieprawidĹ‚owy email lub hasĹ‚o'
    );
  }

  const passwordMatches = await bcrypt.compare(
    password,
    user.password_hash
  );

  if (!passwordMatches) {
    throw createError(
      'INVALID_CREDENTIALS',
      'NieprawidĹ‚owy email lub hasĹ‚o'
    );
  }

  const secret = process.env.JWT_SECRET;
  const expiresIn = process.env.JWT_EXPIRES_IN || '15m';

  if (!secret) {
    throw new Error('Brak konfiguracji JWT_SECRET');
  }

  const accessToken = jwt.sign(
    {
      role: user.role,
    },
    secret,
    {
      algorithm: 'HS256',
      subject: String(user.id),
      expiresIn,
    }
  );

  return {
    accessToken,
    tokenType: 'Bearer',
    expiresIn,
  };
}

async function getMe(userId) {
  const user = await usersRepository.findById(userId);

  if (!user) {
    throw createError('USER_NOT_FOUND', 'UĹĽytkownik nie istnieje');
  }

  return toUserResponse(user);
}

module.exports = {
  register,
  login,
  getMe,
};

    async findPage(filters) {
        const {
            roomId,
            buildingId,
            userId,
            status,
            from,
            to,
            page = 1,
            limit = 20,
            sort = 'startTime'
        } = filters;

        const sortColumns = {
            startTime: 'r.start_time',
            createdAt: 'r.created_at'
        };

        const descending = sort.startsWith('-');
        const sortName = descending ? sort.slice(1) : sort;

        // W aktualnym schemacie nie ma kolumny participants.
        if (!sortColumns[sortName]) {
            throw new ValidationError(
                'Sortowanie po participants wymaga zdefiniowania danych uczestnikĂłw'
            );
        }

        const applyFilters = (query) => {
            if (roomId !== undefined) {
                query.where('r.room_id', roomId);
            }

            if (buildingId !== undefined) {
                query.where('rm.building_id', buildingId);
            }

            if (userId !== undefined) {
                query.where('r.user_id', userId);
            }

            if (status !== undefined) {
                query.where('r.status', status);
            }

            // Rezerwacja musi koĹ„czyÄ‡ siÄ™ po poczÄ…tku zakresu.
            if (from) {
                query.where('r.end_time', '>', new Date(from));
            }

            // Rezerwacja musi zaczynaÄ‡ siÄ™ przed koĹ„cem zakresu.
            if (to) {
                query.where('r.start_time', '<', new Date(to));
            }

            return query;
        };

        const countQuery = applyFilters(
            db('reservations as r')
                .join('rooms as rm', 'r.room_id', 'rm.id')
                .join('users as u', 'r.user_id', 'u.id')
        ).countDistinct({ total: 'r.id' }).first();

        const dataQuery = applyFilters(
            db('reservations as r')
                .join('rooms as rm', 'r.room_id', 'rm.id')
                .join('users as u', 'r.user_id', 'u.id')
        )
            .select(
                'r.id',
                'r.room_id',
                'r.title',
                'r.start_time',
                'r.end_time',
                'r.status',
                'r.created_at',
                'r.updated_at',
                'u.id as user_id',
                'u.email as user_email'
            )
            .orderBy(sortColumns[sortName], descending ? 'desc' : 'asc')
            .orderBy('r.id', 'asc')
            .limit(limit)
            .offset((page - 1) * limit);

        // WyĹ›wietlamy zapytania SQL i przekazywane parametry.
        console.log('\n--- SQL: COUNT ---');
        console.log(countQuery.toSQL());

        console.log('\n--- SQL: DATA ---');
        console.log(dataQuery.toSQL());

        const [countResult, rows] = await Promise.all([
            countQuery,
            dataQuery
        ]);

        const total = Number(countResult?.total ?? 0);

        return {
            data: rows.map(mapReservation),
            meta: {
                page,
                limit,
                total,
                totalPages: Math.ceil(total / limit)
            }
        };
    }

    async createWithRoomLock(reservationData) {
        try {
            return await db.transaction(async (trx) => {
                const roomId =
                    reservationData.roomId ?? reservationData.room_id;

                // Blokujemy salÄ™ na czas wykonywania transakcji.
                const room = await trx('rooms')
                    .where({ id: roomId })
                    .forUpdate()
                    .first();

                if (!room) {
                    throw new NotFoundError('Room not found');
                }

                const startTime =
                    reservationData.startTime ?? reservationData.start_time;
                const endTime =
                    reservationData.endTime ?? reservationData.end_time;

                // Sprawdzamy, czy istnieje kolizja.
                const conflict = await trx('reservations')
                    .where('room_id', roomId)
                    .where('status', 'CONFIRMED')
                    .andWhere('start_time', '<', endTime)
                    .andWhere('end_time', '>', startTime)
                    .first();

                if (conflict) {
                    throw new ConflictError(
                        'Room is already booked for this time period',
                        'ROOM_ALREADY_BOOKED'
                    );
                }

                const [row] = await trx('reservations')
                    .insert({
                        room_id: roomId,
                        user_id: reservationData.userId,
                        title: reservationData.title,
                        start_time: startTime,
                        end_time: endTime,
                        status: 'CONFIRMED'
                    })
                    .returning('*');

                const user = await trx('users')
                    .select('id', 'email')
                    .where({ id: row.user_id })
                    .first();

                return mapReservation({
                    ...row,
                    user_id: user.id,
                    user_email: user.email
                });
            });
        } catch (error) {
            if (error.code === '23P01') {
                throw new ConflictError(
                    'Room is already booked for this time period',
                    'ROOM_ALREADY_BOOKED'
                );
            }

            throw error;
        }
    }
}

const ConflictError = require('../../errors/conflict.error');
const AppError = require('../../errors/app.error');
const NotFoundError = require('../../errors/not-found.error');

class ReservationsService {
  constructor(reservationsRepository, roomsRepository, now = () => new Date()) {
    this.reservationsRepository = reservationsRepository;
    this.roomsRepository = roomsRepository;
    this.now = now;
  }

  async getAllReservations(query = {}, currentUser) {
    if (currentUser.role === 'USER' && query.userId !== undefined) {
      throw new AppError(
        'USER cannot filter reservations by userId',
        403,
        'FORBIDDEN'
      );
    }

    const filters = { ...query };

    if (currentUser.role === 'USER') {
      filters.userId = Number(currentUser.id);
    }

    return this.reservationsRepository.findPage(filters);
  }

  async getReservationById(id, currentUser) {
    const reservation =
      await this.reservationsRepository.findByIdWithRoomAndBuilding(id);

    if (
      !reservation ||
      (
        currentUser.role !== 'ADMIN' &&
        Number(reservation.userId) !== Number(currentUser.id)
      )
    ) {
      throw new NotFoundError('Reservation not found');
    }

    return reservation;
  }

  async createReservation(data, userId) {
    const room = await this.roomsRepository.findById(data.roomId);

    if (!room) {
      throw new NotFoundError('Room not found');
    }

    const startTime = new Date(data.startTime);
    const endTime = new Date(data.endTime);

    if (
      Number.isNaN(startTime.getTime()) ||
      Number.isNaN(endTime.getTime()) ||
      startTime >= endTime
    ) {
      throw new AppError(
        'Invalid reservation time range',
        400,
        'VALIDATION_ERROR'
      );
    }

    if (startTime <= this.now()) {
      throw new AppError(
        'Reservation must start in the future',
        400,
        'VALIDATION_ERROR'
      );
    }

    return this.reservationsRepository.createWithRoomLock({
      roomId: data.roomId,
      userId: Number(userId),
      title: data.title,
      startTime,
      endTime
    });
  }

  async cancelReservation(id, currentUser) {
    const reservation = await this.reservationsRepository.findById(id);

    if (
      !reservation ||
      (
        currentUser.role !== 'ADMIN' &&
        Number(reservation.userId) !== Number(currentUser.id)
      )
    ) {
      throw new NotFoundError('Reservation not found');
    }

    if (reservation.status === 'CANCELLED') {
      return reservation;
    }

    if (reservation.status !== 'CONFIRMED') {
      throw new ConflictError(
        'Only confirmed reservations can be cancelled',
        'INVALID_STATE_TRANSITION'
      );
    }

    if (new Date(reservation.startTime).getTime() <= this.now().getTime()) {
      throw new ConflictError(
        'Cannot cancel a reservation that has already started',
        'RESERVATION_ALREADY_STARTED'
      );
    }

    return this.reservationsRepository.updateStatus(id, 'CANCELLED');
  }
}

module.exports = ReservationsService;
